const dateInput=document.getElementById("resDate");
const dateAvailability=document.getElementById("dateAvailability");
const blockedDateForm=document.getElementById("blockedDateForm");
const blockedDateInput=document.getElementById("blockedDateInput");
const blockedDateReason=document.getElementById("blockedDateReason");
const blockedDateList=document.getElementById("blockedDateList");
const blockedDateCount=document.getElementById("blockedDateCount");

function todayISO(){
  const d=new Date();
  const y=d.getFullYear();
  const m=String(d.getMonth()+1).padStart(2,"0");
  const day=String(d.getDate()).padStart(2,"0");
  return `${y}-${m}-${day}`;
}

function minDate(){
  const t=todayISO();
  dateInput.min=t;
  blockedDateInput.min=t;
}

function getBlockedDates(){
  return read(K.blockedDates)
    .filter(x=>x&&x.date)
    .sort((a,b)=>a.date.localeCompare(b.date));
}

function getReservationByDate(date){
  return allReservations().find(r=>r.date===date);
}

function getDateConflict(date){
  if(!date)return null;

  const blocked=getBlockedDates().find(x=>x.date===date);

  if(blocked){
    return {
      type:"blocked",
      reason:blocked.reason||""
    };
  }

  const reservation=getReservationByDate(date);

  if(reservation){
    return {
      type:"reservation",
      reservation
    };
  }

  return null;
}

function isDateBlocked(date){
  return Boolean(getDateConflict(date));
}

function updateDateAvailability(){
  const date=dateInput.value;

  dateAvailability.className="date-status";
  dateAvailability.textContent="";

  if(!date)return;

  const conflict=getDateConflict(date);

  if(conflict){
    dateAvailability.classList.add("no");

    if(conflict.type==="reservation"){
      dateAvailability.textContent=
        "Fecha no disponible: ya existe una reservacion para ese dia.";
    }else{
      dateAvailability.textContent=
        "Fecha no disponible" +
        (conflict.reason ? ": " + conflict.reason : ". Elige otra fecha.");
    }

    return;
  }

  dateAvailability.classList.add("ok");
  dateAvailability.textContent="Fecha disponible para reservar.";
}

dateInput.addEventListener("change",updateDateAvailability);

function renderBlockedDates(){
  const arr=getBlockedDates();

  blockedDateCount.textContent=arr.length;

  if(!arr.length){
    blockedDateList.innerHTML=
      '<div class="empty">No hay fechas bloqueadas.</div>';
    return;
  }

  blockedDateList.innerHTML=arr.map(x=>`
    <div class="blocked-date-row">
      <div>
        <strong>${esc(fmt(x.date))}</strong>
        <span>${esc(x.reason||"Sin motivo")}</span>
      </div>

      <button
        type="button"
        class="btn btn-red btn-sm del-blocked-date"
        data-date="${esc(x.date)}"
      >
        Liberar
      </button>
    </div>
  `).join("");

  blockedDateList
    .querySelectorAll(".del-blocked-date")
    .forEach(b=>{
      b.onclick=()=>{
        write(
          K.blockedDates,
          getBlockedDates().filter(x=>x.date!==b.dataset.date)
        );

        renderBlockedDates();
        updateDateAvailability();
      };
    });
}

blockedDateForm.onsubmit=e=>{
  e.preventDefault();

  const date=blockedDateInput.value;

  if(!date)return;

  const arr=getBlockedDates();

  if(arr.some(x=>x.date===date)){
    alert("Esa fecha ya esta bloqueada.");
    return;
  }

  if(getReservationByDate(date)){
    alert(
      "Esa fecha ya tiene una reservacion. No necesitas bloquearla manualmente."
    );
    return;
  }

  arr.push({
    date,
    reason:blockedDateReason.value.trim(),
    createdAt:new Date().toISOString()
  });

  write(K.blockedDates,arr);

  e.target.reset();

  minDate();
  renderBlockedDates();
  updateDateAvailability();
};

minDate();

window.addEventListener("storage",event=>{
  if(
    event.key===K.reservations ||
    event.key===K.blockedDates
  ){
    updateDateAvailability();
    renderReservations();
    renderBlockedDates();
  }
});

document.getElementById("reservationForm").onsubmit=e=>{
  e.preventDefault();

  const selectedDate=dateInput.value;

  if(isDateBlocked(selectedDate)){
    updateDateAvailability();
    alert(
      "La fecha seleccionada no esta disponible. Elige otra fecha."
    );
    dateInput.focus();
    return;
  }

  const guests=getGuests().map(g=>({
    id:uid("G"),
    ...g
  }));

  if(guests.length===0){
    alert("Indica cuantas personas van a asistir.");
    guestQuantityInput.focus();
    return;
  }

  if(guests.some(g=>!g.name)){
    alert("Debes escribir el nombre de todos los invitados.");
    return;
  }

  if(guests.length>100){
    alert("La capacidad maxima es de 100 personas.");
    return;
  }

  if(isDateBlocked(selectedDate)){
    updateDateAvailability();
    alert(
      "Esa fecha acaba de dejar de estar disponible. Elige otra fecha."
    );
    dateInput.focus();
    return;
  }

  const r={
    id:uid("HF"),
    name:resName.value.trim(),
    phone:resPhone.value.trim(),
    email:resEmail.value.trim(),
    date:selectedDate,
    type:resType.value,
    packageId:resPackage.value,
    notes:resNotes.value.trim(),
    guests,
    status:"pendiente",
    createdAt:new Date().toISOString()
  };

  const arr=allReservations();
  arr.unshift(r);
  saveReservations(arr);

  confirmText.textContent=
    `${r.name}, recibimos tu reservacion para ${fmt(r.date)} ` +
    `con ${r.guests.length} invitado(s). ` +
    `Esa fecha quedo bloqueada en este sistema. Numero: ${r.id}`;

  confirmBox.classList.add("show");

  e.target.reset();

  guestQuantityInput.value="";

  guestList.innerHTML=
    '<div class="guest-list-empty" id="guestListEmpty">' +
    'Indica primero la cantidad de personas.' +
    '</div>';

  updateGuestSummary();
  minDate();

  dateAvailability.className="date-status";
  dateAvailability.textContent="";

  confirmBox.scrollIntoView({
    behavior:"smooth",
    block:"center"
  });
};

deleteReservation.onclick=()=>{
  if(
    !selectedId ||
    !confirm("Seguro que deseas eliminar esta reservacion completa?")
  ){
    return;
  }

  saveReservations(
    allReservations().filter(r=>r.id!==selectedId)
  );

  selectedId=null;

  renderReservations();
  updateDateAvailability();
};
